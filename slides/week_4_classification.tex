\section{Follow One Request Through Four Methods}

\begin{frame}[fragile]{1. Regex Rules: Match Text Patterns}
  \small A regular expression (regex) specifies a text pattern. Here, \texttt{|} means ``or'': match \texttt{messagerie} or \texttt{email}.
\begin{lstlisting}[language=Python]
import re

def email_rule(query):
    if re.search("messagerie|email", query.lower()):
        return "check_email"
    return None  # No matching rule; no label selected.

email_rule("Ouvrir ma messagerie")  # "check_email"
\end{lstlisting}
  \small The complete classifier has rules for all five labels and an explicit policy when several rules match or none does.
  \begin{block}{A request the rule misses}
    ``Consulter mes e-mails'' returns \texttt{None}: the pattern \texttt{email} does not match \texttt{e-mail}.
  \end{block}
\end{frame}

\begin{frame}{2. Embeddings: Represent the Whole Request by a Vector}
  \small In Week 1, vectors represented text. Here we use a pretrained \textbf{sentence-embedding model}: it converts a request into one fixed-length vector.
  \begin{center}
    \begin{tikzpicture}[node distance=0.4cm]
      \node[routebox,text width=3.4cm,fill=AMU_lt2!50] (query) {``Ouvrir ma messagerie''};
      \node[routebox,text width=3.6cm,right=of query] (embed) {Pretrained\\embedding model};
      \node[routebox,text width=3.4cm,right=of embed,fill=AMU_dk2!10] (vector) {$\mathbf{x}\in\mathbb{R}^{384}$\\384 numbers};
      \draw[routearrow] (query) -- (embed);
      \draw[routearrow] (embed) -- (vector);
    \end{tikzpicture}
  \end{center}
  \small In the lab we use \href{https://huggingface.co/sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2}{multilingual MiniLM}. It supports French and produces 384 numbers per request.
  \par\vspace{0.6em}
  The vector has no intent label yet. We must learn a mapping from these features to the five labels.
\end{frame}

\begin{frame}{Train a Classifier on Those Vectors}
  \small Pair each training request's vector with its reference label. Fit a \textbf{multinomial logistic-regression classifier} to predict those labels.
  \par\vspace{0.4em}
  For a new vector $\mathbf{x}$, calculate one score for each class $c$:
  \[
    s_c = \mathbf{w}_c^\top\mathbf{x}+b_c,
    \qquad \widehat{y}=\underset{c}{\operatorname{argmax}}\;s_c.
  \]
  \small $\mathbf{x}$ is the request's vector; $\mathbf{w}_c$ is the class's weight vector; $b_c$ is its bias. Training learns the weights and biases. $\widehat{y}$ is the predicted label: the class with the largest score.
  \begin{center}
    \begin{tikzpicture}[node distance=0.45cm]
      \node[routebox,text width=3.1cm] (vector) {Request vector\\$\mathbf{x}$};
      \node[routebox,text width=4.2cm,right=of vector] (classifier) {Trained classifier\\Five class scores};
      \node[routebox,text width=3.1cm,right=of classifier,fill=AMU_dk2!10] (label) {Predicted label\\e.g.\ \texttt{check\_email}};
      \draw[routearrow] (vector) -- (classifier);
      \draw[routearrow] (classifier) -- (label);
    \end{tikzpicture}
  \end{center}
  \small We keep the embedding model fixed.
\end{frame}

\begin{frame}{3. Pretrained Classifier}
  \small We supply the request and a description of each of our five labels.
  \par\vspace{0.4em}
  \renewcommand{\arraystretch}{1.1}
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.2cm}>{\raggedright\arraybackslash}p{8.1cm}@{}}
    \toprule
    \textbf{Our label} & \textbf{Description supplied to the model} \\
    \midrule
    \texttt{get\_schedule} & View class timetable \\
    \texttt{check\_email} & Read student email \\
    \texttt{register\_classes} & Register for classes \\
    \texttt{get\_student\_card} & Manage student card \\
    \texttt{find\_library\_info} & Find library information \\
    \bottomrule
  \end{tabular}
  \par\vspace{0.3em}
  \small Our model is multilingual mDeBERTa, already trained to judge whether a text supports a proposed statement.
  \par\vspace{0.4em}
  For example: does ``Ouvrir ma messagerie'' support ``The student wants to read student email''?
  \par\vspace{0.4em}
  The model scores this support for each intent. Select the label with the highest score.
  \note{This is the original notebook's zero-shot-classification pipeline using MoritzLaurer/mDeBERTa-v3-base-mnli-xnli. The base model is trained for natural language inference (NLI): classify a text pair as entailment, contradiction or neutral. The pipeline pairs the request with a statement built from each candidate description, then normalizes entailment logits across candidates in single-label mode. The statement on the slide illustrates the comparison; the exact wording is determined by the pipeline's hypothesis template. These internal relationships are not additional AMU intent labels and need not be a separate lecture topic. Not every pretrained classifier accepts arbitrary candidate descriptions. Sources: https://huggingface.co/docs/transformers/en/main_classes/pipelines and https://huggingface.co/MoritzLaurer/mDeBERTa-v3-base-mnli-xnli.}
\end{frame}

\section{Measure Quality, Time and Cost}

\begin{frame}{Accuracy: How Many Requests Were Classified Correctly?}
  \[
    \text{Accuracy}=\frac{\text{number of correct labels}}{\text{number of test requests}}.
  \]
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\linewidth}
      \small\textbf{Illustrative counts, not model results}
      \par\vspace{0.4em}
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{@{}lr@{}}
        \toprule
        Correct label & 43 \\
        Wrong allowed label & 5 \\
        Invalid reply or no match & 1 \\
        API failure & 1 \\
        \midrule
        Total test requests & 50 \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.47\linewidth}
      \small
      In this example, accuracy is $43/50=86\%$.
      \par\vspace{0.7em}
      The two requests without usable labels stay in the denominator. We also report their failure types separately.
      \par\vspace{0.7em}
      If retries are allowed, fix the policy before testing and include their time and API usage.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Confusion Matrix Shows Which Labels Are Confused}
  \small A \textbf{confusion matrix} counts reference labels against predicted labels. Rows below are the reference labels; columns are the predicted labels.
  \par\vspace{0.6em}
  \begin{center}
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}lrrrrrr@{}}
    \toprule
    \textbf{Expected $\backslash$ predicted} & Schedule & Email & Register & Card & Library & No label \\
    \midrule
    Schedule & \textcolor{AMU_dk2}{8} & 0 & 1 & 0 & 0 & 1 \\
    Email    & 0 & \textcolor{AMU_dk2}{9} & 0 & 1 & 0 & 0 \\
    Register & 0 & 0 & \textcolor{AMU_dk2}{9} & 0 & 0 & 1 \\
    Card     & 0 & 0 & 0 & \textcolor{AMU_dk2}{9} & 1 & 0 \\
    Library  & 1 & 0 & 0 & 1 & \textcolor{AMU_dk2}{8} & 0 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \small Same illustrative 50 requests: the diagonal contains 43 correct labels. ``No label'' combines the invalid reply and API failure.
  \par\vspace{0.3em}
  Which errors send students to the wrong service?
\end{frame}

\begin{frame}{Precision and Recall Answer Different Questions}
  \small Focus on the \textbf{library} class in the same matrix. There are 10 actual library requests; 8 are classified correctly. One card request is also classified as library.
  \par\vspace{0.7em}
  \[
    \text{Library recall}
    =\frac{\text{correct library predictions}}{\text{all actual library requests}}
    =\frac{8}{10}=80\%.
  \]
  \small Of the students who need the library, how many do we send there?
  \par\vspace{0.5em}
  \[
    \text{Library precision}
    =\frac{\text{correct library predictions}}{\text{all library predictions}}
    =\frac{8}{9}\approx89\%.
  \]
  \small Of the students sent to the library, how many actually need it?
  \par\vspace{0.7em}
  Report these measures for every class. Overall accuracy can hide poor performance on a less frequent service.
\end{frame}

\begin{frame}{Measure Prediction Time and Token Usage}
  \begin{itemize}
    \item \textbf{Prediction time}: how long it takes to obtain a label for a request. Start with the model already loaded; exclude training time.
    \item \textbf{API token usage}: record the input and output tokens reported by the API for each request.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Improve a Regex Rule with an LLM}
  \small Our rule misses ``Consulter mes e-mails''. Its reference label is \texttt{check\_email}.
  \begin{block}{Ask the LLM to extend the rule}
    \small Our Python regex is \texttt{messagerie|email}. It misses ``Consulter mes e-mails''. Modify it to match both ``email'' and ``e-mail'', while keeping ``messagerie''.
  \end{block}
  \small A possible revision:
\begin{lstlisting}[language=Python]
pattern = "messagerie|e-?mail"
re.search(pattern, "Consulter mes e-mails".lower())
\end{lstlisting}
  \small \texttt{-?} means the hyphen is optional: both spellings now match.
  \par\vspace{0.6em}
  Run the revised classifier on all development requests. Count corrected predictions and any new errors before keeping the change.
\end{frame}

\begin{frame}[fragile]{Generate Synthetic Debugging Data with an LLM}
  \small Synthetic data are invented examples. Ask an LLM to create ten requests for our development set:
\begin{lstlisting}
Create 10 French student requests, two per label:
get_schedule: view class timetable
check_email: read student email
register_classes: register for classes
get_student_card: manage student card
find_library_info: find library information
Vary the wording; include abbreviations and typos.
Each request must express one intent.
Return CSV with columns text,label.
\end{lstlisting}
  \begin{enumerate}
    \item Read each request and check its assigned label.
    \item Run the classifier on these requests and inspect the mistakes.
    \item Revise the rules or prompt, then rerun the development set.
  \end{enumerate}
\end{frame}

\begin{frame}{Lab: Compare the Four Methods and Defend a Choice}
  \begin{enumerate}
    \item Inspect the requests and agree what each of the five labels means.
    \item Prepare separate training, development and final test examples.
    \item Run the four classifiers. Inspect at least one complete prediction from each method.
